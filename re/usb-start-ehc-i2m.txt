void Titania3_series_start_ehc(uintptr_t UTMI_base, uintptr_t USBC_base, uintptr_t UHC_base, unsigned int flag)
{
printk("Titania3_series_start_ehc start\n");
WW(0x0001,(UTMI_base+0x0*2));
MIU_select_setting_ehc(USBC_base);
WW(0x0C2F,(UTMI_base+0x8*2));
if (flag & 0x2)
{
WW(0x2084,(UTMI_base+0x2*2));
WW(0x8051,(UTMI_base+0x20*2));
}
WB(RB(UTMI_base+0x0*2) | (0x80|0x40|0x02),(UTMI_base+0x0*2));
WB(RB(UTMI_base+0x52*2) | (0x20|0x10| 0x08|0x04|0x02|0x01),(UTMI_base+0x52*2)); })
;
WB(0x0a,(USBC_base));
WB(0x28,(USBC_base));
if (flag & 0x4)
{
if ((flag & 0xF0000000) == 0x10000000)
{
WB(RB(UTMI_base+0x20*2) | 0x76,(UTMI_base+0x20*2));
}
else if ((flag & 0xF0000000) == 0x20000000)
{
WB(RB(UTMI_base+0x20*2) | 0x8e,(UTMI_base+0x20*2));
}
WB(RB(UTMI_base+0x2c*2) | 0x01,(UTMI_base+0x2c*2));
}
if((0x62) & (0x08|0x04|0x02|0x01))
{
WB(((0x62) & (0x08|0x04|0x02|0x01)),(UTMI_base+0x2a*2));
printk("[USB] init squelch level 0x%x\n", RB(UTMI_base+0x2a*2));
}
WB(RB(UTMI_base+0x3c*2) | 0x01,(UTMI_base+0x3c*2));
mdelay(1);}));
WB(RB(UTMI_base+0x3c*2) & (u8)(~0x01),(UTMI_base+0x3c*2));
while ((RB(UTMI_base+0x3c*2) & 0x02) == 0);
if ((0xFFF0 == (RW(UTMI_base+0x3C*2) & 0xFFF0 )) ||
(0x0000 == (RW(UTMI_base+0x3C*2) & 0xFFF0 )) )
printk("WARNING: CA Fail !! \n");
if (flag & 0x1)
WB(RB(UTMI_base+0x0b*2-1) | 0x20,(UTMI_base+0x0b*2-1));
WB((u8)((RB(USBC_base+0x02*2) & ~0x02) | 0x01),(USBC_base+0x02*2));
WB(RB(UHC_base+0x40*2) & (u8)(~0x10),(UHC_base+0x40*2));
udelay(1);
WB(RB(UHC_base+0x40*2) | 0x08,(UHC_base+0x40*2));
WB(RB(UTMI_base+0x0*2) & (u8)(~(0x80|0x40|0x02)),(UTMI_base+0x0*2));
WB(RB(UTMI_base+0x52*2) & (u8)(~(0x20|0x10| 0x08|0x04|0x02|0x01)),(UTMI_base+0x52*2)); })
;
WB(RB(UHC_base+0x81*2-1) | (0x01 | 0x02 | 0x04 | 0x08 | 0x80),(UHC_base+0x81*2-1));
WB((u8)((RB(UTMI_base+0x06*2) & ~0x20) | 0x40),(UTMI_base+0x06*2));
WB((u8)((RB(UTMI_base+0x03*2-1) & ~0x10) | (0x08 | 0x20)),(UTMI_base+0x03*2-1));
WB(RB(UTMI_base+0x07*2-1) & (u8)(~0x02),(UTMI_base+0x07*2-1));
WB(RB(UTMI_base+0x08*2) | 0x08,(UTMI_base+0x08*2));
WB(RB(UTMI_base+0x09*2-1) | (0x01 | 0x80),(UTMI_base+0x09*2-1));
if ((flag & 0x4)==0)
WB(RB(UTMI_base+0x0b*2-1) | 0x80,(UTMI_base+0x0b*2-1));
WB(RB(UTMI_base+0x15*2-1) | 0x20,(UTMI_base+0x15*2-1));
WB(RB(UTMI_base+0x15*2-1) | 0x40,(UTMI_base+0x15*2-1));
WB(RB(UTMI_base+0x40*2) & (u8)(~0x10),(UTMI_base+0x40*2));
WB((0x62),(UTMI_base+0x2a*2));
WB(RB(UTMI_base+0x2c*2) | (0x10),(UTMI_base+0x2c*2));
WB(RB(UTMI_base+0x2d*2-1) | (0x02),(UTMI_base+0x2d*2-1));
WB(RB(UTMI_base+0x2e*2) | (0x00),(UTMI_base+0x2e*2));
WB(RB(UTMI_base+0x2f*2-1) | (0x81),(UTMI_base+0x2f*2-1));
WB(({ u8 __v = ({ u8 __r = __raw_readb((UTMI_base+(0x04*2))); __r; __asm__ __volatile__ ("dsb " "" : : : "memory"); __v; }) | 0x40,(UTMI_base+(0x04*2)));
WB(RB(USBC_base+0x02*2) | 0x40,(USBC_base+0x02*2));
WB(({ u8 __v = ({ u8 __r = __raw_readb((UTMI_base+(0x04*2))); __r; __asm__ __volatile__ ("dsb " "" : : : "memory"); __v; }) | 0x20,(UTMI_base+(0x04*2)));
WB(RB(USBC_base+0x11*2-1) | 0x02,(USBC_base+0x11*2-1));
WB(RB(USBC_base+0x0a*2) | 0x40,(USBC_base+0x0a*2));
WB(RB(USBC_base+0x0b*2-1) & ~(0x02|0x04|0x08|0x10),(USBC_base+0x0b*2-1));
WB(RB(UTMI_base+0x3f*2-1) | 0x80,(UTMI_base+0x3f*2-1));
WB(RB(USBC_base+0x0f*2-1) | 0x08,(USBC_base+0x0f*2-1));
WB(RB(USBC_base+0x10*2) | 0x01,(USBC_base+0x10*2));
WB(RB(USBC_base+0x10*2) | 0x02,(USBC_base+0x10*2));
WB(RB(UTMI_base+0x03*2-1) | 0x04,(UTMI_base+0x03*2-1));
WB(RB(UHC_base+0x34*2) | 0x80,(UHC_base+0x34*2));
WB(RB(UHC_base+0x83*2-1) | 0x10,(UHC_base+0x83*2-1));
WB(RB(UHC_base+0x8C*2) | 0x01,(UHC_base+0x8C*2));
WB(RB(USBC_base+0x13*2-1) | 0x01,(USBC_base+0x13*2-1));
WB(RB(USBC_base+0x20*2) | 0x01,(USBC_base+0x20*2));
WB(RB(USBC_base+0x20*2) | 0x02,(USBC_base+0x20*2));
WB(RB(USBC_base+0x20*2) | 0x04,(USBC_base+0x20*2));
WB(0x05,(USBC_base+0x03*2-1));
if (flag & 0x2)
{
WW(0x0600,(UTMI_base+0x14*2));
WW(0x0078,(UTMI_base+0x10*2));
WW(0x0bfe,(UTMI_base+0x32*2));
}
